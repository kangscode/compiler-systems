cmake_minimum_required(VERSION 3.16)
project(loop_optimization_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(loop_opt_bench src/bench.cpp)
target_compile_options(loop_opt_bench PRIVATE -O3 -march=native)

enable_testing()
add_test(NAME correctness COMMAND loop_opt_bench --n 64 --tile 16 --repeats 1 --check-only)
